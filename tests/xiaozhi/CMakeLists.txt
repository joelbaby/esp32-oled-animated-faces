cmake_minimum_required(VERSION 3.16)
project(standard_adapter_test LANGUAGES C CXX ASM)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${XIAOZHI}/managed_components/lvgl__lvgl" lvgl)
add_executable(adapter_test adapter_test.cc "${PRODUCTION}/oled_display.cc")
target_include_directories(adapter_test PRIVATE "${PRODUCTION}" "${CMAKE_CURRENT_LIST_DIR}/stubs" "${FACES}/include")
target_compile_definitions(adapter_test PRIVATE BUILTIN_TEXT_FONT=lv_font_montserrat_14 BUILTIN_ICON_FONT=lv_font_montserrat_14)
target_compile_features(adapter_test PRIVATE cxx_std_17)
target_compile_options(adapter_test PRIVATE -Wall -Wextra -Wno-unused-parameter -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(adapter_test PRIVATE -fsanitize=undefined)
target_link_libraries(adapter_test PRIVATE lvgl::lvgl m)
